\section{Getting started}

\begin{frame}{Start a deck: two lines}
  \begin{itemize}
    \item Put \texttt{beamerthemelucid.sty} next to your \texttt{.tex} file
  \end{itemize}
  \howto{start}
  \begin{itemize}
    \item Use \texttt{aspectratio=169} for widescreen
  \end{itemize}
  \citeline{Compile twice. The outline slides need a second pass. \quad snippets/start.tex}
\end{frame}

\begin{frame}[allowframebreaks]{The title slide}
  \begin{itemize}
    \item \term{No name, institution or logo appears unless you set one}
    \item \texttt{[plain]} drops the footer, which a title slide should not have
  \end{itemize}
  \howto{title-slide}
  \citeline{snippets/title-slide.tex}
\end{frame}

\begin{frame}[allowframebreaks]{Make a new slide}
  \begin{itemize}
    \item A slide is a \term{frame}. Title in braces, content between
    \item Leave the braces off for a slide with no title
  \end{itemize}
  \howto{new-slide}
  \citeline{snippets/new-slide.tex}
\end{frame}

\begin{frame}[allowframebreaks]{Hide a slide}
  \begin{itemize}
    \item Comment it out to drop it for now
    \item Move it below \texttt{backupframes} to keep it without counting it
  \end{itemize}
  \howto{hide-slide}
  \citeline{snippets/hide-slide.tex}
\end{frame}

\begin{frame}[allowframebreaks]{Speaker notes}
  \begin{itemize}
    \item \texttt{note} sits inside the frame it belongs to
    \item Nothing is shown unless you ask for it, so notes are safe to leave in
  \end{itemize}
  \howto{notes}
  \speakernote{This is a real speaker note. Compile with
    notes=pages and it gets its own page after this slide.}
  \citeline{snippets/notes.tex}
\end{frame}

\begin{frame}{Animation: one slide, several pages}
  \begin{steps}
    \item Each step prints as \term{another page} of the PDF
    \item Nothing moves --- the reader sees a page at a time
    \item So an animated deck has more pages than slides
  \end{steps}
  \citeline{This slide is three pages. In handout mode it collapses back to one.}
\end{frame}

\begin{frame}[allowframebreaks]{Animation: the syntax}
  \howto{animation}
  \citeline{snippets/animation.tex}
\end{frame}

\begin{hidden}
\begin{frame}{You are not seeing this}
  \begin{itemize}
    \item This frame sits inside a \texttt{hidden} block
    \item Compile with \texttt{hidden=show} and it appears
  \end{itemize}
\end{frame}
\end{hidden}

\begin{frame}[allowframebreaks]{Printing an animated deck}
  \begin{itemize}
    \item \term{handout} collapses every step back to one page per slide
    \item Build three files from one body: the talk, the handout, your script
  \end{itemize}
  \howto{handout}
  \citeline{snippets/handout.tex}

\section{Text}

\begin{frame}{Bullets: two levels, and a third to avoid}
  \begin{itemize}
    \item Level one
      \begin{itemize}
        \item Level two
          \begin{itemize}
            \item Level three is muted on purpose
          \end{itemize}
      \end{itemize}
  \end{itemize}
  \howto{bullets}
  \citeline{Aim for two top-level bullets and 25--30 words a slide. \quad snippets/bullets.tex}
\end{frame}

\begin{frame}{Emphasis: three devices, and none of them is bold}
  \begin{itemize}
    \item A \term{primary term}, a \altterm{secondary term}, something \hl{highlighted}
    \item Each one survives greyscale printing on its own
  \end{itemize}
  \howto{emphasis}
  \citeline{snippets/emphasis.tex}
\end{frame}

\begin{frame}[allowframebreaks]{Maths beyond one equation}
  \begin{itemize}
    \item Aligned lines, cases, matrices, and numbering when you need it
  \end{itemize}
  \begin{align*}
    \mathcal{L} &= \mathbb{E}_{\pi}\left[ \sum_t \gamma^t r_t \right] \\
                &= \sum_s d^{\pi}(s)\, v^{\pi}(s)
  \end{align*}
  \[ r(s) = \begin{cases}
       +1 & \text{host recovered} \\
       -1 & \text{host compromised} \\
        0 & \text{otherwise}
     \end{cases} \]
  \howto{math}
  \citeline{snippets/math.tex}
\end{frame}

\begin{frame}{Statements}
  \begin{definition}
    A \term{world model} predicts the next observation from the last.
  \end{definition}
  \begin{theorem}[Nash, 1951]
    Every finite game has an equilibrium in mixed strategies.
  \end{theorem}
  \begin{itemize}
    \item Also: \texttt{lemma}, \texttt{corollary}, \texttt{proof},
      \texttt{example}, \texttt{block}
  \end{itemize}
  \citeline{They take the deck's colours, and their bodies stay upright to read from the back.}
\end{frame}

\begin{frame}{Pointing at one part of an equation}
  \[ q_j = \frac{e^{\tint{accent}{\lambda}\, \mathrm{EU}_j(x)}}
                {\sum_i e^{\lambda\, \mathrm{EU}_i(x)}} \]
  \begin{itemize}
    \item Colour the piece you are talking about, or reveal it with
      \texttt{emphat} and \texttt{showfrom}
  \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Colour, highlight, underline}
  \begin{itemize}
    \item \term{a primary term}, \altterm{a secondary term}, \hl{highlighted},
      \ul{underlined}, \alert{alerted}
    \item Or reach into the palette directly: \tint{gold}{gold},
      \tint{muted}{muted}, \tint{termB}{teal}
  \end{itemize}
  \howto{color}
  \citeline{snippets/color.tex \quad Three or four colours carry 95\% of the text
    in a good deck.}
\end{frame}

\begin{frame}[allowframebreaks]{Citing sources}
